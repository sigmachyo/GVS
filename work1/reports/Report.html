4. Фрагменты исходного кода

Data.cuh (фрагмент)

template <typename AtomT>
class Data {
private:
    int size_;
    AtomT* data_;
public:
    explicit Data(int size) : size_(size), data_(nullptr) {
        if (size_ > 0) cudaMalloc(&data_, size_ * sizeof(AtomT));
    }
    ~Data() { if (data_) cudaFree(data_); }
    // конструкторы копирования, перемещения, операторы = (Rule of Five)
};

VectorView.cuh

template <typename AtomT>
class VectorView {
private:
    AtomT* data_;
    int size_;
public:
    __host__ __device__ VectorView(AtomT* data, int size) : data_(data), size_(size) {}
    __host__ __device__ int size() const { return size_; }
    __host__ __device__ AtomT& operator[](int n) { return data_[n]; }
};

kernel_vecadd.cuh

template <typename AtomT>
__global__ void kernel_vecadd(VectorView<AtomT> lhs, VectorView<AtomT> rhs, VectorView<AtomT> out) {
    int i = blockIdx.x * blockDim.x + threadIdx.x;
    if (i < out.size()) {
        out[i] = lhs[i] + rhs[i];
    }
}
